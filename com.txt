<#
.SYNOPSIS
    Discover SQL Server instances, pick one interactively, authenticate,
    list its databases, and generate (optionally run) the matching
    generate_db_map.py command line.

.DESCRIPTION
    Combines two discovery methods (sqlcmd -L broadcast + registry scan),
    presents a numbered menu, handles Windows or SQL auth via Get-Credential
    (no plaintext password on screen), live-tests the connection, lists
    databases, lets you pick one, then builds and optionally runs:

        python generate_db_map.py --server <server> --database <db> [--auth-type ... --username ...]

.NOTES
    Requires: sqlcmd (SQL Server command line tools) and the SqlServer
    PowerShell module (or falls back to sqlcmd for the connection test
    if the module isn't installed).
#>

# ---------------------------------------------------------------------------
# 1. DISCOVERY — sqlcmd -L (network broadcast) + registry (local instances)
# ---------------------------------------------------------------------------

function Get-SqlCmdInstances {
    Write-Host "Scanning network for SQL Server instances (sqlcmd -L)..." -ForegroundColor Cyan
    $instances = @()
    try {
        $raw = sqlcmd -L 2>$null
        foreach ($line in $raw) {
            $line = $line.Trim()
            # sqlcmd -L output lines look like: "MACHINENAME\INSTANCENAME" or "MACHINENAME"
            if ($line -and $line -notmatch "Servers:" -and $line -notmatch "^-+$") {
                $instances += $line
            }
        }
    }
    catch {
        Write-Host "  sqlcmd -L failed or is unavailable: $($_.Exception.Message)" -ForegroundColor Yellow
    }
    return $instances | Sort-Object -Unique
}

function Get-RegistryInstances {
    Write-Host "Scanning registry for locally-installed SQL Server instances..." -ForegroundColor Cyan
    $instances = @()

    $regPaths = @(
        "HKLM:\SOFTWARE\Microsoft\Microsoft SQL Server\Instance Names\SQL",
        "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Microsoft SQL Server\Instance Names\SQL"
    )

    foreach ($path in $regPaths) {
        if (Test-Path $path) {
            try {
                $props = Get-ItemProperty -Path $path -ErrorAction Stop
                foreach ($prop in $props.PSObject.Properties) {
                    # Skip PowerShell's own metadata properties
                    if ($prop.Name -notmatch '^PS') {
                        $instanceName = $prop.Name
                        if ($instanceName -eq "MSSQLSERVER") {
                            $instances += $env:COMPUTERNAME
                        }
                        else {
                            $instances += "$($env:COMPUTERNAME)\$instanceName"
                        }
                    }
                }
            }
            catch {
                # Path existed but read failed; ignore and continue
            }
        }
    }

    return $instances | Sort-Object -Unique
}

function Get-AllSqlInstances {
    $fromSqlCmd  = Get-SqlCmdInstances
    $fromRegistry = Get-RegistryInstances

    $combined = @($fromSqlCmd + $fromRegistry) | Where-Object { $_ } | Sort-Object -Unique
    return $combined
}

# ---------------------------------------------------------------------------
# 2. MENU — let the user pick a server
# ---------------------------------------------------------------------------

function Select-SqlInstance {
    param([string[]]$Instances)

    if ($Instances.Count -eq 0) {
        Write-Host "No instances found automatically." -ForegroundColor Yellow
        $manual = Read-Host "Enter a SQL Server instance name manually (e.g. SERVER\INSTANCE)"
        return $manual
    }

    Write-Host "`nDiscovered SQL Server instances:" -ForegroundColor Green
    for ($i = 0; $i -lt $Instances.Count; $i++) {
        Write-Host "  [$($i + 1)] $($Instances[$i])"
    }
    Write-Host "  [$($Instances.Count + 1)] Enter manually"

    do {
        $choice = Read-Host "`nSelect a server by number"
        $valid = ($choice -as [int]) -and ([int]$choice -ge 1) -and ([int]$choice -le ($Instances.Count + 1))
        if (-not $valid) { Write-Host "Invalid choice, try again." -ForegroundColor Red }
    } while (-not $valid)

    $choiceIndex = [int]$choice
    if ($choiceIndex -eq ($Instances.Count + 1)) {
        return Read-Host "Enter the instance name manually"
    }
    return $Instances[$choiceIndex - 1]
}

# ---------------------------------------------------------------------------
# 3. AUTH — Windows or SQL auth, credential collected securely
# ---------------------------------------------------------------------------

function Get-SqlAuthChoice {
    Write-Host "`nAuthentication type:" -ForegroundColor Green
    Write-Host "  [1] Windows Authentication (current user)"
    Write-Host "  [2] SQL Server Authentication (username/password)"

    do {
        $choice = Read-Host "Select auth type"
    } while ($choice -notin @("1", "2"))

    if ($choice -eq "1") {
        return @{ AuthType = "windows"; Credential = $null }
    }
    else {
        # Get-Credential never echoes the password to the console
        $cred = Get-Credential -Message "Enter SQL Server login credentials"
        return @{ AuthType = "sql"; Credential = $cred }
    }
}

# ---------------------------------------------------------------------------
# 4. CONNECTION TEST + DATABASE LISTING
# ---------------------------------------------------------------------------

function Test-SqlConnectionAndListDatabases {
    param(
        [string]$ServerInstance,
        [hashtable]$AuthInfo
    )

    $query = "SELECT name FROM sys.databases WHERE database_id > 4 ORDER BY name;"  # skip system DBs

    try {
        if (Get-Module -ListAvailable -Name SqlServer) {
            Import-Module SqlServer -ErrorAction Stop

            if ($AuthInfo.AuthType -eq "windows") {
                $result = Invoke-Sqlcmd -ServerInstance $ServerInstance -Query $query -ErrorAction Stop
            }
            else {
                $plainUser = $AuthInfo.Credential.UserName
                $plainPass = $AuthInfo.Credential.GetNetworkCredential().Password
                $result = Invoke-Sqlcmd -ServerInstance $ServerInstance -Query $query `
                    -Username $plainUser -Password $plainPass -ErrorAction Stop
            }
            $databases = $result | ForEach-Object { $_.name }
        }
        else {
            # Fallback: raw sqlcmd.exe if the SqlServer module isn't installed
            if ($AuthInfo.AuthType -eq "windows") {
                $raw = sqlcmd -S $ServerInstance -E -Q $query -h -1 2>&1
            }
            else {
                $plainUser = $AuthInfo.Credential.UserName
                $plainPass = $AuthInfo.Credential.GetNetworkCredential().Password
                $raw = sqlcmd -S $ServerInstance -U $plainUser -P $plainPass -Q $query -h -1 2>&1
            }
            if ($LASTEXITCODE -ne 0) { throw "sqlcmd returned exit code $LASTEXITCODE : $raw" }
            $databases = $raw | ForEach-Object { $_.Trim() } | Where-Object { $_ -and $_ -notmatch "rows affected" }
        }

        return @{ Success = $true; Databases = $databases }
    }
    catch {
        return @{ Success = $false; Error = $_.Exception.Message }
    }
}

# ---------------------------------------------------------------------------
# 5. COMMAND GENERATION + OPTIONAL EXECUTION
# ---------------------------------------------------------------------------

function New-GenerateDbMapCommand {
    param(
        [string]$ServerInstance,
        [string]$Database,
        [hashtable]$AuthInfo
    )

    if ($AuthInfo.AuthType -eq "windows") {
        $cmd = "python generate_db_map.py --server `"$ServerInstance`" --database `"$Database`" --auth-type windows"
    }
    else {
        $user = $AuthInfo.Credential.UserName
        $cmd = "python generate_db_map.py --server `"$ServerInstance`" --database `"$Database`" --auth-type sql --username `"$user`" --password `"<hidden>`""
    }
    return $cmd
}

# ---------------------------------------------------------------------------
# MAIN
# ---------------------------------------------------------------------------

function Main {
    $instances = Get-AllSqlInstances
    $server = Select-SqlInstance -Instances $instances

    $authInfo = Get-SqlAuthChoice

    Write-Host "`nTesting connection to '$server'..." -ForegroundColor Cyan
    $test = Test-SqlConnectionAndListDatabases -ServerInstance $server -AuthInfo $authInfo

    if (-not $test.Success) {
        Write-Host "Connection failed: $($test.Error)" -ForegroundColor Red
        return
    }

    if ($test.Databases.Count -eq 0) {
        Write-Host "Connected successfully, but no user databases were found." -ForegroundColor Yellow
        return
    }

    Write-Host "`nDatabases on '$server':" -ForegroundColor Green
    for ($i = 0; $i -lt $test.Databases.Count; $i++) {
        Write-Host "  [$($i + 1)] $($test.Databases[$i])"
    }

    do {
        $dbChoice = Read-Host "`nSelect a database by number"
        $valid = ($dbChoice -as [int]) -and ([int]$dbChoice -ge 1) -and ([int]$dbChoice -le $test.Databases.Count)
        if (-not $valid) { Write-Host "Invalid choice, try again." -ForegroundColor Red }
    } while (-not $valid)

    $database = $test.Databases[[int]$dbChoice - 1]

    $command = New-GenerateDbMapCommand -ServerInstance $server -Database $database -AuthInfo $authInfo

    Write-Host "`nGenerated command:" -ForegroundColor Green
    Write-Host "  $command`n" -ForegroundColor White

    if ($authInfo.AuthType -eq "sql") {
        Write-Host "Note: the printed command shows '<hidden>' for the password." -ForegroundColor Yellow
        Write-Host "If you run it now, the real password is substituted automatically." -ForegroundColor Yellow
    }

    $runNow = Read-Host "Run this command now? (y/N)"
    if ($runNow -match '^[Yy]') {
        if ($authInfo.AuthType -eq "sql") {
            $plainUser = $authInfo.Credential.UserName
            $plainPass = $authInfo.Credential.GetNetworkCredential().Password
            $execCommand = "python generate_db_map.py --server `"$server`" --database `"$database`" --auth-type sql --username `"$plainUser`" --password `"$plainPass`""
        }
        else {
            $execCommand = "python generate_db_map.py --server `"$server`" --database `"$database`" --auth-type windows"
        }
        Write-Host "`nRunning..." -ForegroundColor Cyan
        Invoke-Expression $execCommand
    }
    else {
        Write-Host "Skipped execution. Copy the command above when you're ready to run it." -ForegroundColor Cyan
    }
}

Main
